\noindent
\textbf{GLOSARIUM TEKNIS}

\subsection*{A}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[API (\textit{Application Programming Interface}):] Sekumpulan aturan dan protokol yang memungkinkan aplikasi \textit{frontend} berkomunikasi dengan \textit{server backend} untuk pertukaran data secara otomatis.
\end{description}

\subsection*{B}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Backend:] Bagian dari sistem yang menangani logika bisnis, pemrosesan data, dan komunikasi dengan \textit{database} yang tidak terlihat langsung oleh pengguna.
\end{description}

\subsection*{C}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Code Coverage:] Metrik yang menunjukkan persentase baris kode program yang telah berhasil dieksekusi selama proses pengujian unit dijalankan.
\end{description}

\subsection*{D}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Database (Supabase):] Kumpulan data terorganisir yang disimpan secara digital dalam tabel (Pelanggan, Layanan, Transaksi) untuk mengelola operasional laundry.
\end{description}

\subsection*{G}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Go (Golang):] Bahasa pemrograman yang digunakan untuk membangun \textit{backend} sistem karena efisiensi dan performanya dalam menangani \textit{concurrency}.
\end{description}

\subsection*{J}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[JWT (\textit{JSON Web Token}):] Standar industri untuk pembuatan token akses yang digunakan dalam proses otentikasi keamanan saat admin atau kasir melakukan login.
\end{description}

\subsection*{U}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Unit Testing:] Metode pengujian di mana komponen terkecil dari kode diuji secara mandiri untuk memastikan fungsinya berjalan sesuai logika yang dirancang.
\end{description}

\subsection*{W}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Webhook:] Mekanisme yang digunakan oleh \textit{payment gateway} untuk mengirimkan notifikasi data secara otomatis ke \textit{server} saat transaksi pembayaran berhasil dideteksi.
\end{description}

\vspace{0.5cm}
\noindent
\textbf{GLOSARIUM NON-TEKNIS}

\subsection*{A}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Analisis Fungsional:] Proses identifikasi kebutuhan utama sistem yang mencakup fitur-fitur inti seperti transaksi, manajemen pelanggan, dan pelaporan.
    \item[Analisis Non-Fungsional:] Analisis terhadap aspek pendukung sistem agar beroperasi optimal, mencakup keamanan data, kinerja, dan kenyamanan pengguna (\textit{usability}).
\end{description}

\subsection*{D}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Dashboard:] Halaman utama aplikasi yang menyajikan ringkasan statistik bisnis seperti total omset dan kinerja operasional secara visual.
\end{description}

\subsection*{F}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Flowchart:] Diagram yang merepresentasikan alur kerja atau algoritma proses bisnis laundry, mulai dari penerimaan order hingga penyerahan cucian.
\end{description}

\subsection*{P}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Point of Sale (POS):] Sistem digital yang digunakan untuk mengelola transaksi penjualan, pencatatan pesanan, dan pengelolaan stok secara terpadu.
\end{description}

\subsection*{Q}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[QRIS (\textit{Quick Response Code Indonesian Standard}):] Standar kode QR nasional untuk pembayaran digital yang memudahkan pelanggan melakukan transaksi melalui dompet digital.
\end{description}

\subsection*{R}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Real-time:] Kondisi pemrosesan data yang terjadi seketika, seperti pembaruan status pembayaran yang langsung terdeteksi oleh sistem tanpa perlu pengecekan manual.
\end{description}

\subsection*{S}
\begin{description}[leftmargin=0pt, labelfont=\bfseries]
    \item[Struk Transaksi:] Bukti pembayaran digital atau fisik yang berisi rincian layanan laundry, berat cucian, dan total biaya yang harus dibayar pelanggan.
\end{description}
